cmake_minimum_required(VERSION 3.16)
project(luxi_bringup)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(luxi_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(luxi_core REQUIRED)

install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
 find_package(OpenSSL REQUIRED)
 find_package(SQLite3 REQUIRED)
 find_package(rosbag2_cpp REQUIRED)
 add_executable(integration_probe test/integration_probe.cpp)
 target_include_directories(integration_probe PRIVATE ../arm_motion/include ../data_recorder/include "${LUXI_ROOT}/3part/external/mujoco/include")
 target_link_libraries(integration_probe PkgConfig::HTTPLIB OpenSSL::Crypto SQLite::SQLite3 "${LUXI_ROOT}/3part/external/mujoco/lib/libmujoco.so.3.12.0")
 ament_target_dependencies(integration_probe rclcpp luxi_core luxi_interfaces std_msgs rosbag2_cpp)
 install(TARGETS integration_probe DESTINATION lib/${PROJECT_NAME})
 add_test(NAME integration_capture COMMAND bash "${LUXI_ROOT}/tool/integration.sh")
 set_tests_properties(integration_capture PROPERTIES TIMEOUT 90)
endif()
ament_package()
