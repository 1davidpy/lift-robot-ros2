cmake_minimum_required(VERSION 3.16)
project(luxi_act_io)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(luxi_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
add_executable(act_rgb_http src/rgb_http.cpp)
target_link_libraries(act_rgb_http PkgConfig::HTTPLIB nlohmann_json::nlohmann_json)
ament_target_dependencies(act_rgb_http rclcpp luxi_interfaces)
add_executable(act_action_udp src/action_udp.cpp)
target_link_libraries(act_action_udp nlohmann_json::nlohmann_json)
ament_target_dependencies(act_action_udp rclcpp std_msgs)
install(TARGETS act_rgb_http act_action_udp DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME act_io_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/integration.py
    $<TARGET_FILE:act_rgb_http> $<TARGET_FILE:act_action_udp>)
  set_tests_properties(act_io_integration PROPERTIES TIMEOUT 60)
endif()
ament_package()
