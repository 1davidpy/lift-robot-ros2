cmake_minimum_required(VERSION 3.16)
project(luxi_quest_receiver)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(luxi_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(luxi_core REQUIRED)

add_executable(quest_receiver src/quest_receiver.cpp)
ament_target_dependencies(quest_receiver rclcpp luxi_interfaces luxi_core)
install(TARGETS quest_receiver DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME latest_input
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_latest_input.py $<TARGET_FILE:quest_receiver>)
  set_tests_properties(latest_input PROPERTIES TIMEOUT 30)
endif()
ament_package()
