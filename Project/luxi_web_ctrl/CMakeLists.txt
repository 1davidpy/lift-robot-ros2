cmake_minimum_required(VERSION 3.16)
project(luxi_web_ctrl)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(luxi_core REQUIRED)
find_package(luxi_interfaces REQUIRED)
find_package(luxi_data_recorder REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
add_executable(web_ctrl src/web_ctrl.cpp)
target_link_libraries(web_ctrl PkgConfig::HTTPLIB OpenSSL::Crypto)
ament_target_dependencies(web_ctrl ament_index_cpp rclcpp luxi_core luxi_interfaces luxi_data_recorder std_msgs)
install(TARGETS web_ctrl DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY web DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 add_executable(http_probe test/http_probe.cpp)
 target_link_libraries(http_probe PkgConfig::HTTPLIB)
 ament_target_dependencies(http_probe rclcpp luxi_core luxi_interfaces std_msgs)
 install(TARGETS http_probe DESTINATION lib/${PROJECT_NAME})
 add_test(NAME web_gateway COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/test/run_http_test.sh" "${LUXI_ROOT}")
 set_tests_properties(web_gateway PROPERTIES TIMEOUT 45)
endif()
ament_package()
